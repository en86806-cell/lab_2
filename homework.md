## Условие задачи
Определить путь, который проедут санки с горки высотой h метров, углом наклона A радиан и коэффициентом трения k.

## 2. Реализация программы
```c
#include <stdio.h>
#include <locale.h>
#include <math.h>

int main() {
    setlocale(LC_ALL, "RUS");

    
    float h = 10.0;  
    float A = 0.5;   
    float k = 0.1;   
    float g = 9.8;   

    
    float S1 = h / sin(A); 
    float v_squared = 2 * g * h * (1 - k / tan(A)); 

    // Проверка, поедут ли санки
    if (v_squared < 0) {
        printf("Санки не поедут! Сила трения слишком велика.\n");
        return 1;
    }

    float v = sqrt(v_squared); 
    float S2 = v_squared / (2 * g * k); 
    float S_total = S1 + S2; 

    
    
    printf("РАСЧЁТ ПУТИ ДВИЖЕНИЯ САНОК \n");
    

    printf("УСЛОВИЯ:\n");
    printf("- Высота горки (h): %.2f м\n", h);
    printf("- Угол наклона (A): %.2f рад\n", A);
    printf("- Коэффициент трения (k): %.2f\n", k);
    printf("- Ускорение свободного падения (g): %.2f м/с^2\n", g);

    printf("\n---------------------------------------\n\n");

    printf("РАСЧЁТ:\n");
    printf("- Длина склона горки (S1) = h / sin(A) = %.2f м.\n", S1);
    printf("- Скорость у подножия (v) = %.2f м/с.\n", v);
    printf("- Тормозной путь по горизонтали (S2) = v^2 / (2 * g * k) = %.2f м.\n", S2);

    printf("\n---------------------------------------\n\n");

    printf("ИТОГОВЫЙ ОБЩИЙ ПУТЬ САНОК: %.2f м.\n\n", S_total);

    printf("---------------------------------------\n");
    printf("Информация о разработчике:\n");
    printf("Никулина Елена, группа [ИЦТ-262]\n"); 
    printf("---------------------------------------\n");

    return 0;
}
